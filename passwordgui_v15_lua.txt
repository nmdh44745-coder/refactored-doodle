local function loadData()
    local data = { 
        trialUsed = false, 
        expiresAt = 0, 
        permanent = false,
        autoWinEnabled = false,
        wallClipEnabled = false,
        espEnabled = false,
        floatEnabled = false,
        aimbotEnabled = false
    }

    if type(isfile) == "function" and type(readfile) == "function" then
        local ok, raw = pcall(function()
            if isfile(SAVE_FILE) then
                return readfile(SAVE_FILE)
            end
        end)

        if ok and type(raw) == "string" and raw ~= "" then
            local decodeOK, decoded = pcall(function()
                return game:GetService("HttpService"):JSONDecode(raw)
            end)
            if decodeOK and type(decoded) == "table" then
                data.trialUsed = decoded.trialUsed == true
                data.expiresAt = tonumber(decoded.expiresAt) or 0
                data.permanent = decoded.permanent == true
                data.autoWinEnabled = decoded.autoWinEnabled == true
                data.wallClipEnabled = decoded.wallClipEnabled == true
                data.espEnabled = decoded.espEnabled == true
                data.floatEnabled = decoded.floatEnabled == true
                data.aimbotEnabled = decoded.aimbotEnabled == true
            end
        end
    end

    return data
end

-- Universal Dragging Helper Function (Touch + Mouse Support)
--------------------------------------------------------------------------------
local function makeDraggable(frame, handle)
    handle = handle or frame
    local dragging, dragStart, startPosition, dragInput = false, nil, nil, nil

    handle.InputBegan:Connect(function(inputObject)
        if inputObject.UserInputType == Enum.UserInputType.MouseButton1
            or inputObject.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = inputObject.Position
            startPosition = frame.Position

            inputObject.Changed:Connect(function()
                if inputObject.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    handle.InputChanged:Connect(function(inputObject)
        if inputObject.UserInputType == Enum.UserInputType.MouseMovement
            or inputObject.UserInputType == Enum.UserInputType.Touch then
            dragInput = inputObject
        end
    end)

    UserInputService.InputChanged:Connect(function(inputObject)
        if dragging and inputObject == dragInput then
            local delta = inputObject.Position - dragStart
            frame.Position = UDim2.new(
                startPosition.X.Scale, startPosition.X.Offset + delta.X,
                startPosition.Y.Scale, startPosition.Y.Offset + delta.Y
            )
        end
    end)
end

local espHighlights = {}
local espLines = {}

local function clearESP()
    for _, hl in pairs(espHighlights) do
        if hl then hl:Destroy() end
    end
    table.clear(espHighlights)

    for _, line in pairs(espLines) do
        if line then
            pcall(function() line:Remove() end)
            pcall(function() line:Destroy() end)
        end
    end
    table.clear(espLines)
end

RunService.Heartbeat:Connect(function()
    local char = LocalPlayer.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end

    -- 1. Wall Clip
    if wallClipState then
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") and part.CanCollide then
                part.CanCollide = false
            end
        end
    end

    -- 2. Float Walk
    if floatState then
        if root.Velocity.Y < 0 then
            root.Velocity = Vector3.new(root.Velocity.X, 0, root.Velocity.Z)
        end
    end

    -- 3. Auto Win: Teleports to nearest player EXCEPT whitelisted ones
    if autoWinState then
        local myPos = root.Position
        local bestTargetHead = nil
        local shortestDist = math.huge

        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and not whitelistedPlayers[player.Name] and player.Character then
                local eChar = player.Character
                local eHead = eChar:FindFirstChild("Head")
                local eHum = eChar:FindFirstChildOfClass("Humanoid")

                if eHead and eHum and eHum.Health > 0 then
                    local dist = (eHead.Position - myPos).Magnitude
                    if dist < shortestDist then
                        shortestDist = dist
                        bestTargetHead = eHead
                    end
                end
            end
        end

        if bestTargetHead then
            root.CFrame = CFrame.new(bestTargetHead.Position + Vector3.new(0, 20, 0))
            root.Velocity = Vector3.zero
        end
    end

    -- 4. Aimbot (Aims screen at nearest enemy head under 10 radius)
    if aimbotState then
        local myPos = root.Position
        local closestEnemyHead = nil
        local shortestDist = 10

        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                local eChar = player.Character
                local eHead = eChar:FindFirstChild("Head")
                local eHum = eChar:FindFirstChildOfClass("Humanoid")

                if eHead and eHum and eHum.Health > 0 then
                    local dist = (eHead.Position - myPos).Magnitude
                    if dist <= shortestDist then
                        shortestDist = dist
                        closestEnemyHead = eHead
                    end
                end
            end
        end

        if closestEnemyHead then
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, closestEnemyHead.Position)
        end
    end

    -- 5. ESP
    if espState then
        local screenSize = Camera.ViewportSize
        local lineStart = Vector2.new(screenSize.X / 2, screenSize.Y)

        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                local eChar = player.Character
                local eRoot = eChar:FindFirstChild("HumanoidRootPart")
                local eHead = eChar:FindFirstChild("Head")
                local eHum = eChar:FindFirstChildOfClass("Humanoid")

                if eRoot and eHead and eHum and eHum.Health > 0 then
                    local dist = (eRoot.Position - root.Position).Magnitude
                    if dist <= 1000 then
                        local hl = espHighlights[player]
                        if not hl or hl.Parent ~= eChar then
                            if hl then hl:Destroy() end
                            hl = Instance.new("Highlight")
                            hl.Name = "G15_ESP"
                            hl.Adornee = eChar
                            hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                            hl.FillColor = Color3.fromRGB(255, 0, 0)
                            hl.FillTransparency = 0.4
                            hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                            hl.OutlineTransparency = 0
                            hl.Parent = eChar
                            espHighlights[player] = hl
                        end

                        local headScreenPos, onScreen = Camera:WorldToViewportPoint(eHead.Position)
                        if onScreen then
                            local line = espLines[player]
                            if not line and Drawing then
                                pcall(function()
                                    line = Drawing.new("Line")
                                    line.Thickness = 1.5
                                    line.Color = Color3.fromRGB(255, 0, 0)
                                    line.Transparency = 1
                                    espLines[player] = line
                                end)
                            end

                            if line then
                                line.From = lineStart
                                line.To = Vector2.new(headScreenPos.X, headScreenPos.Y)
                                line.Visible = true
                            end
                        else
                            if espLines[player] then
                                espLines[player].Visible = false
                            end
                        end
                    else
                        if espHighlights[player] then
                            espHighlights[player]:Destroy()
                            espHighlights[player] = nil
                        end
                        if espLines[player] then
                            pcall(function() espLines[player]:Remove() end)
                            espLines[player] = nil
                        end
                    end
                else
                    if espHighlights[player] then
                        espHighlights[player]:Destroy()
                        espHighlights[player] = nil
                    end
                    if espLines[player] then
                        pcall(function() espLines[player]:Remove() end)
                        espLines[player] = nil
                    end
                end
            end
        end
    else
        clearESP()
    end
end)

--------------------------------------------------------------------------------
-- Authentication & Timer Flow
--------------------------------------------------------------------------------
local function setStatus(text, color)
    statusLbl.Text = text
    statusLbl.TextColor3 = color or Color3.fromRGB(255, 110, 110)
end

local timerVersion = 0

local function showUnlockedPermanent()
    loginFrame.Visible = false
    timerWidget.Visible = false
    floatBtn.Visible = true
end

local function showUnlockedTrial(expiresAt)
    timerVersion += 1
    local thisVersion = timerVersion

    loginFrame.Visible = false
    floatBtn.Visible = true
    timerWidget.Visible = true

    task.spawn(function()
        while gui.Parent and timerVersion == thisVersion do
            local remaining = math.max(0, expiresAt - os.time())
            if remaining <= 0 then
                floatBtn.Visible = false
                cheatMenu.Visible = false
                timerWidget.Visible = false
                loginFrame.Visible = true
                keyInput.Text = ""
                keyInput.PlaceholderText = "Key expired — enter new key"
                unlockBtn.Text = "UNLOCK ACCESS"
                subtitle.Text = "Your access key has ended."
                setStatus("✖ KEY EXPIRED", Color3.fromRGB(255, 105, 105))
                return
            end

            local hours = math.floor(remaining / 3600)
            local mins = math.floor((remaining % 3600) / 60)
            local secs = remaining % 60
            timerLbl.Text = string.format("%02d:%02d:%02d", hours, mins, secs)
            task.wait(0.5)
        end
    end)
end

local function applySavedStatus()
    if data.permanent then
        showUnlockedPermanent()
    elseif data.trialUsed and data.expiresAt > os.time() then
        showUnlockedTrial(data.expiresAt)
    elseif data.trialUsed then
        loginFrame.Visible = true
        floatBtn.Visible = false
        timerWidget.Visible = false
        keyInput.PlaceholderText = "Key expired — enter key"
        unlockBtn.Text = "UNLOCK ACCESS"
        subtitle.Text = "Your key duration has ended."
        setStatus("✖ KEY EXPIRED", Color3.fromRGB(255, 105, 105))
    end
end

local busy = false

unlockBtn.Activated:Connect(function()
    if busy then return end

    local key = keyInput.Text
    if key == "" then
        setStatus("✖ PLEASE ENTER A KEY")
        return
    end

    busy = true
    unlockBtn.Text = "CHECKING..."
    setStatus("VERIFYING KEY...", Color3.fromRGB(195, 175, 255))

    if key == PERMANENT_KEY then
        data.permanent = true
        saveData()
        showUnlockedPermanent()

    elseif key == HOUR_KEY then
        data.trialUsed = true
        data.expiresAt = os.time() + HOUR_SECONDS
        saveData()
        showUnlockedTrial(data.expiresAt)

    elseif key == TRIAL_KEY then
        if data.trialUsed then
            keyInput.Text = ""
            keyInput.PlaceholderText = "Key already used"
            setStatus("✖ KEY EXPIRED — ALREADY REDEEMED")
            unlockBtn.Text = "UNLOCK ACCESS"
        else
            data.trialUsed = true
            data.expiresAt = os.time() + TRIAL_SECONDS
            saveData()
            showUnlockedTrial(data.expiresAt)
        end
    else
        setStatus("✖ INVALID KEY")
        unlockBtn.Text = "UNLOCK ACCESS"
    end

    busy = false
end)

    gui:Destroy()
end)
